cmake_minimum_required(VERSION 3.20)
project(TensorCraftTrace LANGUAGES CXX)

# Build with the same BWAPI 4.4.0 headers and Win32 BWAPILIB as Pluto.
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native BWAPI tournament plug-in requires MSVC on Windows.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "BWAPI 4.4.0 / StarCraft 1.16.1 require an x86 Win32 build.")
endif()
set(BWAPI_INCLUDE_DIR "" CACHE PATH "BWAPI 4.4.0 bwapi/include directory")
set(BWAPILIB_LIBRARY "" CACHE FILEPATH "Matching x86 Release BWAPILIB.lib")
if(NOT EXISTS "${BWAPI_INCLUDE_DIR}/BWAPI.h")
  message(FATAL_ERROR "Provide -DBWAPI_INCLUDE_DIR=<BWAPI-4.4.0>/bwapi/include")
endif()
if(NOT EXISTS "${BWAPILIB_LIBRARY}")
  message(FATAL_ERROR "Provide -DBWAPILIB_LIBRARY=<path-to-x86-Release-BWAPILIB.lib>")
endif()
add_library(TensorCraftTrace SHARED TraceTournament.cpp)
target_compile_features(TensorCraftTrace PRIVATE cxx_std_14)
target_compile_definitions(TensorCraftTrace PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(TensorCraftTrace PRIVATE "${BWAPI_INCLUDE_DIR}")
target_link_libraries(TensorCraftTrace PRIVATE "${BWAPILIB_LIBRARY}")
set_target_properties(TensorCraftTrace PROPERTIES PREFIX "" OUTPUT_NAME "TensorCraftTrace")
if(MSVC)
  target_compile_options(TensorCraftTrace PRIVATE /W4 /EHsc)
  target_link_options(TensorCraftTrace PRIVATE /MACHINE:X86)
endif()
